cmake_minimum_required(VERSION 3.18)
project(customer_session_search VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(BOOST_JSON_INCLUDE_DIR boost/json/src.hpp REQUIRED)
add_executable(customer-session-search src/main.cpp)
target_include_directories(customer-session-search PRIVATE ${BOOST_JSON_INCLUDE_DIR})
target_compile_options(customer-session-search PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS customer-session-search RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:customer-session-search>)
endif()
