cmake_minimum_required(VERSION 3.18)
project(bfind VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(BOOST_JSON_INCLUDE_DIR boost/json/src.hpp REQUIRED)
if(NOT EXISTS "${BOOST_JSON_INCLUDE_DIR}/boost/version.hpp")
  message(FATAL_ERROR "Boost version header is required; Boost 1.75.0 or newer is supported")
endif()
file(STRINGS "${BOOST_JSON_INCLUDE_DIR}/boost/version.hpp" BFIND_BOOST_VERSION_LINE
     REGEX "^#define[ \t]+BOOST_VERSION[ \t]+[0-9]+")
string(REGEX MATCH "[0-9]+" BFIND_BOOST_VERSION_NUMBER "${BFIND_BOOST_VERSION_LINE}")
if(NOT BFIND_BOOST_VERSION_NUMBER OR BFIND_BOOST_VERSION_NUMBER LESS 107500)
  message(FATAL_ERROR "Boost 1.75.0 or newer is required")
endif()

include(CheckCXXCompilerFlag)
include(CheckCXXSourceCompiles)
check_cxx_compiler_flag("-fstack-protector-strong" BFIND_HAS_STACK_PROTECTOR)
check_cxx_source_compiles("#include <features.h>\n#ifndef __GLIBC__\n#error not glibc\n#endif\nint main() { return 0; }" BFIND_HAS_GLIBC)
add_library(bfind_hardening INTERFACE)
if(BFIND_HAS_STACK_PROTECTOR)
  target_compile_options(bfind_hardening INTERFACE -fstack-protector-strong)
endif()
if(BFIND_HAS_GLIBC AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bfind_hardening INTERFACE -include "${CMAKE_CURRENT_SOURCE_DIR}/src/hardening.hpp")
endif()
add_library(bfind_dates STATIC src/dates.cpp)
target_include_directories(bfind_dates PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(bfind_dates PRIVATE -Wall -Wextra -Wpedantic)
add_library(bfind_terminal STATIC src/terminal.cpp)
target_include_directories(bfind_terminal PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(bfind_terminal PRIVATE -Wall -Wextra -Wpedantic)
add_executable(bfind src/main.cpp)
add_library(bfind_process STATIC src/process.cpp)
target_include_directories(bfind_process PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(bfind_process PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(bfind PRIVATE bfind_process bfind_terminal bfind_dates)
target_include_directories(bfind PRIVATE ${BOOST_JSON_INCLUDE_DIR})
target_compile_options(bfind PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS bfind RUNTIME DESTINATION bin)
foreach(BFIND_TARGET IN ITEMS bfind bfind_process bfind_terminal bfind_dates)
  target_link_libraries(${BFIND_TARGET} PRIVATE bfind_hardening)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(bfind-fixture src/main.cpp)
  target_include_directories(bfind-fixture PRIVATE ${BOOST_JSON_INCLUDE_DIR})
  target_link_libraries(bfind-fixture PRIVATE bfind_process bfind_terminal bfind_dates)
  target_compile_options(bfind-fixture PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(bfind-fixture PRIVATE BFIND_TEST_SEARCH_ROOT="${CMAKE_CURRENT_BINARY_DIR}/test-notes" BFIND_TEST_EPOCH=1742040000)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:bfind-fixture> ${CMAKE_CURRENT_BINARY_DIR}/test-notes $<TARGET_FILE:bfind> 1742040000)
  add_executable(process-tests tests/process_test.cpp)
  target_link_libraries(process-tests PRIVATE bfind_process bfind_terminal)
  target_compile_options(process-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(terminal-tests tests/terminal_test.cpp)
  target_link_libraries(terminal-tests PRIVATE bfind_terminal)
  target_compile_options(terminal-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(date-tests tests/dates_test.cpp)
  target_link_libraries(date-tests PRIVATE bfind_dates)
  target_compile_options(date-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME date-window COMMAND date-tests)
  add_test(NAME terminal-output COMMAND terminal-tests)
  add_test(NAME process-deadlines COMMAND process-tests)
  set_tests_properties(process-deadlines PROPERTIES TIMEOUT 10)
  foreach(BFIND_TARGET IN ITEMS bfind-fixture process-tests terminal-tests date-tests)
    target_link_libraries(${BFIND_TARGET} PRIVATE bfind_hardening)
  endforeach()
endif()
